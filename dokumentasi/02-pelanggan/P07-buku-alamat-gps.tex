\flowmeta{P07}{Buku Alamat dan GPS}{Pelanggan}

\begin{flowsection}{Tujuan}
Mengelola beberapa alamat pengiriman, lengkap dengan penanda titik lokasi pada
peta agar kurir mudah menemukan alamat.
\end{flowsection}

\begin{flowsection}{Prasyarat}
Sudah login sebagai pelanggan.
\end{flowsection}

\begin{flowsection}{Langkah Mengelola Alamat}
\begin{enumerate}
  \item Buka \texttt{/profile} lalu tab \textbf{Buku Alamat}.
  \item Klik \textbf{Tambah Alamat Baru} (atau Ubah pada alamat yang ada).
  \item Isi label alamat, nama penerima, nomor HP, dan alamat lengkap.
  \item Tentukan titik lokasi pada peta: seret penanda, atau klik
        \textbf{Isi Otomatis (GPS)} untuk memakai lokasi perangkat.
  \item Simpan alamat.
\end{enumerate}
\end{flowsection}

\begin{flowsection}{Hasil yang Diharapkan}
Alamat tersimpan dan dapat dipilih saat checkout; titik koordinat tercatat untuk
keperluan pengiriman.
\end{flowsection}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Field wajib kosong $\rightarrow$ pesan validasi.
  \item Izin lokasi ditolak pada perangkat $\rightarrow$ isi titik peta secara manual.
\end{itemize}
\end{flowsection}

\end{flowsection}
