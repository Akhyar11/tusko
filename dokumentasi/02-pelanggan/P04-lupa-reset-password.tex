\flowmeta{P04}{Lupa dan Reset Password}{Pelanggan}

\begin{flowsection}{Tujuan}
Memulihkan akses akun ketika pelanggan lupa kata sandi, dengan mengirim tautan
reset ke email terdaftar.
\end{flowsection}

\begin{flowsection}{Prasyarat}
Memiliki akun terdaftar dan akses ke email terdaftar.
\end{flowsection}

\begin{flowsection}{Langkah Lupa Kata Sandi}
\begin{enumerate}
  \item Buka halaman \texttt{/forgot-password}.
  \item Masukkan \textbf{alamat email} terdaftar.
  \item Klik tombol kirim.
  \item Sistem menampilkan pesan ``Tautan terkirim''.
  \item Buka email, klik tautan reset, lalu isi kata sandi baru pada halaman
        \texttt{/reset-password}.
\end{enumerate}
\end{flowsection}

\shot{gambar/pelanggan/p04-1.png}{Halaman lupa kata sandi}{fig:p04-1}

\begin{flowsection}{Hasil yang Diharapkan}
Muncul pesan generik ``Jika email tersebut terdaftar, kami telah mengirimkan
tautan reset kata sandi'' (tidak mengungkap apakah email terdaftar atau tidak,
demi keamanan).
\end{flowsection}

\shot{gambar/pelanggan/p04-2.png}{Konfirmasi tautan reset telah dikirim}{fig:p04-2}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Permintaan terlalu sering $\rightarrow$ pesan agar mencoba beberapa saat lagi.
  \item Jika pengiriman email gagal (mis. gangguan server email), sistem tetap
        menampilkan pesan generik dan mencatat kegagalan di log.
  \item Tautan reset kedaluwarsa $\rightarrow$ minta tautan baru.
\end{itemize}
\end{flowsection}

\begin{flowsection}{Kaitan Modul}
FE \texttt{ForgotPasswordPage.jsx}, \texttt{ResetPasswordPage.jsx}. BE
\texttt{AuthController@forgotPassword}, \texttt{@resetPassword}. Rute
\texttt{POST /api/auth/forgot-password}, \texttt{POST /api/auth/reset-password}.
\end{flowsection}
