\flowmeta{P20}{Halaman Legal dan Informasi}{Pelanggan/Tamu}

\begin{flowsection}{Tujuan}
Menyediakan halaman informasi resmi toko: Tentang Kami, Kontak, Syarat \&
Ketentuan, Kebijakan Privasi, Kebijakan Pengembalian, Kebijakan Pengiriman,
dan FAQ.
\end{flowsection}

\begin{flowsection}{Prasyarat}
Tidak perlu login; konten diisi admin melalui Pengaturan Sistem.
\end{flowsection}

\begin{flowsection}{Halaman Tersedia}
\begin{itemize}
  \item \texttt{/tentang} --- Tentang Kami (identitas \& kontak toko).
  \item \texttt{/kontak} --- Hubungi Kami.
  \item \texttt{/syarat-ketentuan} --- Syarat \& Ketentuan.
  \item \texttt{/kebijakan-privasi} --- Kebijakan Privasi.
  \item \texttt{/kebijakan-pengembalian} --- Kebijakan Pengembalian/Retur.
  \item \texttt{/kebijakan-pengiriman} --- Kebijakan Pengiriman.
  \item \texttt{/faq} --- Pertanyaan yang Sering Diajukan.
\end{itemize}
\end{flowsection}

\begin{flowsection}{Hasil yang Diharapkan}
Pengunjung membaca informasi resmi; konten dapat berubah tanpa mengubah kode
(dinamis dari pengaturan admin).
\end{flowsection}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Konten belum diisi admin $\rightarrow$ halaman menampilkan pesan bahwa
        informasi belum diatur (perlu diisi pada Pengaturan Sistem).
\end{itemize}
\end{flowsection}

\begin{flowsection}{Kaitan Modul}
FE \texttt{LegalPage.jsx}. BE \texttt{StoreProfileController},
\texttt{SettingsRegistry} (grup \texttt{store}). Rute \texttt{GET /api/store/profile}.
\end{flowsection}
