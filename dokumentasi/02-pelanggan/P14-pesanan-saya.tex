\flowmeta{P14}{Pesanan Saya}{Pelanggan}

\begin{flowsection}{Tujuan}
Melihat daftar dan detail pesanan pelanggan, memantau status, serta melakukan
tindakan lanjutan (bayar ulang, lacak, retur, ulasan).
\end{flowsection}

\begin{flowsection}{Prasyarat}
Sudah login sebagai pelanggan.
\end{flowsection}

\begin{flowsection}{Langkah Melihat Pesanan}
\begin{enumerate}
  \item Buka halaman \texttt{/orders} (\textbf{Pesanan Saya}).
  \item Daftar pesanan tampil beserta status terkini.
  \item Klik sebuah pesanan untuk membuka \textbf{detail pesanan}.
  \item Dari detail, lakukan tindakan sesuai status: \textbf{Bayar Sekarang}
        (belum dibayar), \textbf{Lacak} (dikirim), \textbf{Ajukan Retur} (selesai),
        atau \textbf{Beri Ulasan}.
\end{enumerate}
\end{flowsection}

\begin{flowsection}{Siklus Pesanan}
Gambar berikut merangkum perjalanan status pesanan dari pembayaran hingga selesai
dan tindakan purnajual.
\end{flowsection}

\shot{gambar/pelanggan/p14-2.png}{Siklus status pesanan pelanggan}{fig:p14-2}

\begin{flowsection}{Hasil yang Diharapkan}
Pelanggan melihat status pesanan yang akurat dan dapat melanjutkan ke tindakan
yang relevan.
\end{flowsection}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Belum ada pesanan $\rightarrow$ tampil empty-state.
  \item Pesanan kedaluwarsa $\rightarrow$ tidak dapat dibayar; buat pesanan baru.
\end{itemize}
\end{flowsection}

\begin{flowsection}{Kaitan Modul}
FE \texttt{OrdersPage.jsx}, \texttt{OrderDetailPage.jsx}, \texttt{orderService.js}.
BE \texttt{OrderController}. Rute \texttt{GET /api/orders} (sisi pelanggan).
\end{flowsection}
