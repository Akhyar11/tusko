\flowmeta{P12}{Pembayaran Midtrans}{Pelanggan}

\begin{flowsection}{Tujuan}
Membayar pesanan secara daring melalui Midtrans \textbf{Core API} dengan
antarmuka penuh milik TUSKO: Virtual Account dan QRIS/e-Wallet. Nomor Virtual
Account / kode QR ditampilkan langsung pada halaman pesanan TUSKO (tanpa popup
pihak ketiga).
\end{flowsection}

\begin{flowsection}{Prasyarat}
Checkout telah lengkap (alamat, pengiriman, metode pembayaran). Kredensial
Midtrans telah diisi admin pada Pengaturan Sistem.
\end{flowsection}

\begin{flowsection}{Langkah Pembayaran}
\begin{enumerate}
  \item Pada checkout, pilih metode pembayaran Midtrans:
        \textbf{Virtual Account} (BCA/Mandiri/BRI/BNI) atau \textbf{QRIS \& E-Wallet}.
  \item Klik \textbf{Bayar Sekarang}.
  \item Sistem membuat transaksi melalui Midtrans Core API dan menampilkan
        \textbf{nomor Virtual Account} atau \textbf{kode QR} pada halaman pesanan.
  \item Pelanggan menyelesaikan pembayaran sesuai instruksi.
  \item Midtrans mengirim notifikasi (webhook); backend memverifikasi
        \texttt{signature\_key} lalu menandai pesanan \textbf{PAID}.
\end{enumerate}
\end{flowsection}

\shot{gambar/pelanggan/p12-1.png}{Memilih metode pembayaran Midtrans (Virtual Account) pada checkout TUSKO}{fig:p12-1}

\begin{flowsection}{Hasil yang Diharapkan}
Halaman pesanan menampilkan nomor Virtual Account / kode QR beserta batas waktu
pembayaran; setelah terbayar status berubah menjadi \textbf{PAID}. Seluruh
tampilan instruksi berada pada UI TUSKO.
\end{flowsection}

\shot{gambar/pelanggan/p12-2.png}{Instruksi pembayaran Virtual Account ditampilkan pada halaman pesanan TUSKO}{fig:p12-2}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Kredensial Midtrans belum diisi $\rightarrow$ muncul pesan kesalahan.
  \item Pembayaran kedaluwarsa $\rightarrow$ pesanan belum terbayar; dapat dibuat ulang.
  \item Webhook gagal diverifikasi $\rightarrow$ status tidak berubah (aman).
\end{itemize}
\end{flowsection}

\begin{flowsection}{Kaitan Modul}
FE \texttt{CheckoutPage.jsx}, \texttt{OrderSuccessPage.jsx}, \texttt{checkoutService.js}
(\texttt{chargeOrder}). BE \texttt{CheckoutController@charge},
\texttt{MidtransService}, \texttt{MidtransWebhookController}. Rute
\texttt{POST /api/orders/\{id\}/charge}. Tabel \texttt{payments},
\texttt{midtrans\_transactions}.
\end{flowsection}
