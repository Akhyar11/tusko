\flowmeta{A05}{Stok dan Mutasi}{Admin}

\begin{flowsection}{Tujuan}
Melihat stok tiap produk dan mencatat mutasi (restok, pengurangan, penyesuaian).
\end{flowsection}

\begin{flowsection}{Prasyarat}
Login sebagai admin dengan peran yang memiliki akses modul ini.
\end{flowsection}

\begin{flowsection}{Langkah}
\begin{enumerate}
  \item Buka Manajemen Stok (/admin/stock).
  \item Gunakan filter untuk mencari produk/SKU.
  \item Menu baris → Restok / Kurangi / Penyesuaian untuk mencatat mutasi stok.
  \item Perubahan stok tersimpan ke server.
\end{enumerate}
\end{flowsection}

\shot{gambar/admin/a05-1.png}{Halaman Stok dan Mutasi pada Panel Admin}{fig:a05-1}

\begin{flowsection}{Hasil yang Diharapkan}
Stok produk diperbarui dan tercatat sebagai mutasi.
\end{flowsection}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Jumlah keluar tidak boleh melebihi stok tersedia.
\end{itemize}
\end{flowsection}

\begin{flowsection}{Kaitan Modul}
\begin{itemize}
  \item \texttt{StockManagementPage.jsx}
  \item \texttt{StockMutationPage.jsx}
  \item \texttt{InventoryController}
  \item \texttt{InventoryService}
  \item \texttt{/api/inventory}
\end{itemize}
\end{flowsection}
