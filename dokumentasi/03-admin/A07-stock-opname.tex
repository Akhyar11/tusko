% A07 — Stock Opname
% ============================================================
\flowmeta{A07}{Stock Opname}{Admin}

\begin{flowsection}{Tujuan}
Melakukan sesi hitung fisik stok dan mencatat penyesuaian selisih.
\end{flowsection}

\begin{flowsection}{Prasyarat}
Login sebagai admin dengan peran yang memiliki akses modul ini.
\end{flowsection}

\begin{flowsection}{Langkah}
\begin{enumerate}
  \item Buka Stock Opname (/admin/stock-opname).
  \item Buat sesi opname baru, lalu pilih \textbf{sumber}:
        \textbf{Seluruh Stok Gudang}, atau \textbf{Berdasarkan Penerimaan (GRN)}
        untuk merekonsiliasi barang dari dokumen penerimaan tertentu.
  \item Setelah sumber dipilih, sistem memuat \textbf{hanya produk/varian yang
        benar-benar memiliki saldo} di gudang tersebut (beserta stok sistem; dan
        jumlah diterima bila sumbernya GRN).
  \item Isi \textbf{stok fisik} per item/varian; selisih = fisik $-$ sistem
        dihitung otomatis.
  \item Simpan sesi (draft), lalu \textbf{Ajukan} untuk proses persetujuan.
  \item \textbf{Setujui \& Sesuaikan Stok} agar saldo stok otoritatif dan jurnal
        penyesuaian dibuat.
\end{enumerate}
\end{flowsection}

\begin{flowsection}{Anti-Manipulasi Stok}
Opname hanya berlaku untuk item yang terdaftar pada stok gudang terpilih.
Produk/varian yang tidak ada di gudang \textbf{ditolak} oleh sistem (validasi
server) sehingga opname tidak dapat dipakai untuk menyuntikkan stok semaunya.
Bila sumbernya dokumen penerimaan (GRN), item juga wajib termasuk dalam GRN
tersebut dan gudangnya harus sama.
\end{flowsection}

\shot{gambar/admin/a07-1.png}{Halaman Stock Opname pada Panel Admin}{fig:a07-1}

\begin{flowsection}{Hasil yang Diharapkan}
Selisih opname diterapkan sebagai penyesuaian stok.
\end{flowsection}

\begin{flowsection}{Penanganan Error dan Kasus Khusus}
\begin{itemize}
  \item Item yang tidak terdaftar di stok gudang terpilih $\rightarrow$ ditolak
        (``Item tidak terdaftar pada stok gudang terpilih'').
  \item Sesi harus disetujui agar penyesuaian stok berlaku.
\end{itemize}
\end{flowsection}

\begin{flowsection}{Kaitan Modul}
\begin{itemize}

  \item \texttt{/api/stock-opnames}, \texttt{/api/stock-opnames/candidates},
        \texttt{/api/stock-opnames/\{id\}/approve}
\end{itemize}
\end{flowsection}
