\begin{flowsection}{Status Pesanan (Umum)}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|l|p{9.5cm}|}
\hline
\textbf{Status} & \textbf{Arti} \\ \hline
Menunggu Pembayaran & Pesanan dibuat, belum dibayar \\ \hline
Dibayar (PAID) & Pembayaran terverifikasi (Midtrans/webhook atau manual) \\ \hline
Diproses & Pesanan diproses gudang \\ \hline
Dikirim & Pengiriman dibuat, resi tersedia \\ \hline
Selesai & Paket diterima; poin loyalitas bertambah \\ \hline
Dibatalkan & Pesanan dibatalkan (kedaluwarsa/ditolak) \\ \hline
Retur / Refund & Proses pengembalian barang/dana \\ \hline
\end{tabular}
\end{center}
Status disimpan pada tabel \texttt{order\_statuses} (dinamis).
\end{flowsection}

\begin{flowsection}{Status Pembayaran}
Menunggu $\rightarrow$ Terbayar $\rightarrow$ (Refund/Dibatalkan). Disimpan pada
\texttt{payment\_statuses}. Metode: Midtrans (VA/QRIS/e-Wallet/Kartu) dan
Transfer Bank Manual.
\end{flowsection}

\begin{flowsection}{Jenis Mutasi Stok}
Masuk (restok), Keluar (rusak/sampel/selisih), Penyesuaian (koreksi opname).
Disimpan pada \texttt{stock\_mutation\_types}.
\end{flowsection}
